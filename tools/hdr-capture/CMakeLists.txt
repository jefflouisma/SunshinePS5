cmake_minimum_required(VERSION 3.25)
project(sunshine_hdr_capture LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0=1.0.29)
add_executable(uvc_capture uvc_capture.c)
target_compile_definitions(uvc_capture PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(uvc_capture PRIVATE -Wall -Wextra -Wpedantic -Werror)
# The root-owned macOS deployment must not load a user-writable Homebrew dylib.
if(APPLE)
  find_file(LIBUSB_ARCHIVE NAMES libusb-1.0.a PATHS ${LIBUSB_LIBRARY_DIRS}
    NO_DEFAULT_PATH REQUIRED)
  find_library(IOKIT_FRAMEWORK IOKit REQUIRED)
  find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
  find_library(SECURITY_FRAMEWORK Security REQUIRED)
  target_include_directories(uvc_capture SYSTEM PRIVATE ${LIBUSB_INCLUDE_DIRS})
  target_link_libraries(uvc_capture PRIVATE ${LIBUSB_ARCHIVE} objc
    ${IOKIT_FRAMEWORK} ${COREFOUNDATION_FRAMEWORK} ${SECURITY_FRAMEWORK})
else()
  target_link_libraries(uvc_capture PRIVATE PkgConfig::LIBUSB)
endif()
include(CTest)
add_test(NAME capture_parser_contract COMMAND uvc_capture --self-test)
set_tests_properties(capture_parser_contract PROPERTIES TIMEOUT 10)
